"use strict"; class Employee { constructor(id, name, department, salary) { this.id = id; this.name = name; this.department = department; this.salary = salary; } getAnnualSalary() { return this.salary * 12; } applyBonus(percent) { this.salary = this.salary + (this.salary * percent / 100); } } const employees = [ new Employee(1, "Amit", "IT", 40000), new Employee(2, "Sara", "HR", 35000), new Employee(3, "Kiran", "Sales", 30000), new Employee(4, "Rahul", "IT", 50000), new Employee(5, "Priya", "Finance", 45000) ]; // Apply 10% bonus to each employee employees.forEach(employee => employee.applyBonus(10)); // Calculate and display annual salary employees.forEach(employee => { console.log(`Name: ${employee.name}`); console.log(`Department: ${employee.department}`); console.log(`Annual Salary: ₹${employee.getAnnualSalary()}`); console.log("-----------------------"); }); // Calculate total annual payout const totalPayout = employees.reduce( (total, employee) => total + employee.getAnnualSalary(), 0 ); console.log(`Total Annual Payout: ₹${totalPayout}`);
